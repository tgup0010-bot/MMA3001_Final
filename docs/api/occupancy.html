128
129    Args:
130        co2: Output of :func:`co2_series`.

132        rolling_window: Trailing bins (including the current one) averaged

136        A tuple ``(table, info)``. ``table`` has a ``time`` column, every
137        name in :data:`NUMERIC_FEATURES_WITH_OCCUPANCY`, and ``target``

142        ValueError: If the two series share no overlapping timestamps.
143    """

146        raise ValueError("co2 and occupancy series have no overlapping, non-missing timestamps.")

149    aligned["co2_lag1"] = aligned["co2"].shift(1)
150    aligned["co2_rolling_1h"] = (

152    )
153    aligned["target"] = aligned["co2"].shift(-1)
154
155    calendar = _add_calendar_features(aligned.index)
156    table = pd.concat([aligned, calendar], axis=1).reset_index(names="time")
157
158    feature_cols = list(NUMERIC_FEATURES_WITH_OCCUPANCY)
159    needed = feature_cols + ["target"]
160
161    n_before = len(table)
162    table = table.dropna(subset=needed).reset_index(drop=True)
163    n_after = len(table)
164
165    info = {
166        "rows_before": n_before,
167        "rows_after": n_after,
168        "rows_dropped": n_before - n_after,
169        "pct_dropped": 100.0 * (n_before - n_after) / n_before if n_before else 0.0,
170    }
171

def building_occupancy_series( occ_resampled: pandas.DataFrame, *, room_col: str = 'floorspaceid', time_col: str = 'collecteddate') -> pandas.Series: 
45def building_occupancy_series(
46    occ_resampled: pd.DataFrame,
47    *,
48    room_col: str = "floorspaceid",
49    time_col: str = "collecteddate",
50) -> pd.Series:
51    """Aggregate per-room occupancy into one building-wide signal.
52
53    Args:
54        occ_resampled: Output of
55            :func:`occupancy.preprocessing.resample_occupancy`, covering
56            all rooms on the same time grid.
57        room_col: Column identifying the room/zone.
58        time_col: Column holding the time bin.
59
60    Returns:
61        A ``pandas.Series`` indexed by time bin, holding the number of
62        zones occupied at that instant (0 to 5). Bins where every room is
63        simultaneously missing (all ``NaN``) are left ``NaN`` rather than
64        counted as zero occupied, since "no data" and "confirmed empty"
65        are different claims.
66    """
67    # pivot_table silently drops any index value whose row is entirely NaN
68    # across every column -- exactly the "all rooms missing" bins this
69    # function needs to detect and preserve. Reindex against every
70    # timestamp actually present in the input to put those bins back.
71    wide = occ_resampled.pivot_table(index=time_col, columns=room_col, values="occupied")
72    all_timestamps = pd.Index(sorted(occ_resampled[time_col].unique()), name=time_col)
73    wide = wide.reindex(all_timestamps)
74    all_missing = wide.isna().all(axis=1)
75    total = wide.sum(axis=1, min_count=1)
76    total[all_missing] = float("nan")
77    return total.rename("building_occupancy")

def trapezoidal_integral( times: pandas.Series | numpy.ndarray, values: pandas.Series | numpy.ndarray) -> float: 
27def trapezoidal_integral(times: pd.Series | np.ndarray, values: pd.Series | np.ndarray) -> float:
28    """Newton-Cotes trapezoidal rule, honouring the true (possibly irregular) node spacing.
29
30    Per Week 6.1/7.1: real sensor timestamps are rarely perfectly evenly
31    spaced, and averaging the gaps and pretending they are equal invents
32    error that has nothing to do with measurement noise. This uses each
33    consecutive pair's *actual* time gap.
34
35    Args:
36        times: Timestamps (any type ``pandas`` can subtract to get a
37            ``Timedelta``, e.g. a tz-aware ``DatetimeIndex`` or column).
38        values: The integrand's value at each timestamp (same length as
39            ``times``), e.g. CO2 concentration in ppm.
40
41    Returns:
42        The accumulated integral, in (value-unit x hours) -- e.g. ppm.h
43        for a CO2 series -- computed as
44        :math:`\\sum_i (t_{i+1}-t_i)(f_i+f_{i+1})/2`.
45
46    Raises:
47        ValueError: If fewer than 2 points are given.
48    """
49    times = pd.to_datetime(pd.Series(times).reset_index(drop=True))
50    values = pd.Series(values).reset_index(drop=True).astype(float)
51    if len(times) < 2:
52        raise ValueError("Need at least 2 (time, value) points to integrate.")
53
54    gaps_hours = times.diff().dt.total_seconds().to_numpy()[1:] / 3600.0
55    f = values.to_numpy()
56    panel_areas = gaps_hours * (f[:-1] + f[1:]) / 2.0
57    return float(panel_areas.sum())

def trapezoidal_integral_gap_aware( times: pandas.Series | numpy.ndarray, values: pandas.Series | numpy.ndarray, max_gap_hours: float) -> dict: 
 60def trapezoidal_integral_gap_aware(
 61    times: pd.Series | np.ndarray,
 62    values: pd.Series | np.ndarray,
 63    max_gap_hours: float,
 64) -> dict:
 65    """Trapezoidal integration that excludes panels spanning a sensor outage.
 66
 67    :func:`trapezoidal_integral` applies the formula literally to whatever
 68    gap exists between consecutive readings -- correct for genuine
 69    irregular sampling, but if two readings are separated by a real sensor
 70    outage (hours or even days with no data), the formula silently implies
 71    the value varied *linearly* across that whole gap, which is not a
 72    measurement, it's an unstated assumption. This mirrors
 73    ``occupancy.preprocessing.resample_occupancy``'s own ``max_gap``
 74    parameter: panels wider than ``max_gap_hours`` are excluded from the
 75    total rather than integrated across, and the excluded duration is
 76    reported so the coverage gap is visible, not silently absorbed.
 77
 78    Args:
 79        times: Timestamps, as in :func:`trapezoidal_integral`.
 80        values: Integrand values, as in :func:`trapezoidal_integral`.
 81        max_gap_hours: The longest inter-reading gap still integrated
 82            across.
 83
 84    Returns:
 85        A dict with ``total`` (the integral over included panels only,
 86        value-unit x hours), ``hours_included``, ``hours_excluded``, and
 87        ``n_panels_excluded``.
 88    """
 89    times = pd.to_datetime(pd.Series(times).reset_index(drop=True))
 90    values = pd.Series(values).reset_index(drop=True).astype(float)
 91    if len(times) < 2:
 92        raise ValueError("Need at least 2 (time, value) points to integrate.")
 93
 94    gaps_hours = times.diff().dt.total_seconds().to_numpy()[1:] / 3600.0
 95    f = values.to_numpy()
 96    panel_areas = gaps_hours * (f[:-1] + f[1:]) / 2.0
 97
 98    included = gaps_hours <= max_gap_hours
 99    return {
100        "total": float(panel_areas[included].sum()),
101        "hours_included": float(gaps_hours[included].sum()),
102        "hours_excluded": float(gaps_hours[~included].sum()),
103        "n_panels_excluded": int((~included).sum()),
104    }

def simpsons_integral(values: pandas.Series | numpy.ndarray, bin_hours: float) -> float: 
107def simpsons_integral(values: pd.Series | np.ndarray, bin_hours: float) -> float:
108    """Composite Simpson's 1/3 rule -- requires equally spaced samples.
109
110    Args:
111        values: The integrand's value at each equally spaced node, e.g.
112            CO2 concentration (ppm) on a regular time grid. Must have an
113            odd length (an even number of intervals) -- Simpson's 1/3
114            rule's classical requirement (Week 6.1).
115        bin_hours: The (constant) spacing between nodes, in hours.
116
117    Returns:
118        The accumulated integral, in (value-unit x hours).
119
120    Raises:
121        ValueError: If fewer than 3 points are given, or the number of
122            intervals is not even.
123    """
124    f = np.asarray(values, dtype=float)
125    n = len(f) - 1
126    if n < 2:
127        raise ValueError("Need at least 3 points (2 intervals) for Simpson's rule.")
128    if n % 2 != 0:
129        raise ValueError(
130            f"Simpson's 1/3 rule needs an even number of intervals, got {n} "
131            "(odd number of points required)."
132        )
133
134    odd_sum = f[1:-1:2].sum()
135    even_sum = f[2:-1:2].sum()
136    return float((bin_hours / 3.0) * (f[0] + 4 * odd_sum + 2 * even_sum + f[-1]))

Returns: The accumulated integral, in (value-unit x hours).

def richardson_order_check(estimate_h: float, estimate_h_half: float, order: int) -> dict: 
139def richardson_order_check(
140    estimate_h: float, estimate_h_half: float, order: int
141) -> dict:
142    """Check whether halving the step reduced error by the expected factor.
143
144    Per Week 6.2: halving ``h`` should reduce a method of order ``p``'s
145    leading error by about ``2**p`` -- 4x for trapezoidal (p=2), 16x for
146    Simpson (p=4). Since the true integral is unknown here (real sensor
147    data, not a test problem with a known answer), this reports the
148    *change* between the two estimates as convergence evidence, exactly
149    as Week 6.2 describes -- not a claim of the true error.
150
151    Args:
152        estimate_h: The integral estimate at step size ``h``.
153        estimate_h_half: The estimate at step size ``h/2``.
154        order: The method's assumed order of accuracy (2 for trapezoidal,
155            4 for Simpson).
156
157    Returns:
158        A dict with ``change`` (the raw difference between estimates),
159        ``pct_change`` (relative to the finer estimate), and
160        ``expected_reduction_factor`` (``2**order``, for comparison against
161        how much the change itself shrank relative to a still-finer pair,
162        if available).
163    """
164    change = estimate_h_half - estimate_h
165    pct_change = 100.0 * change / estimate_h_half if estimate_h_half else float("nan")
166    return {
167        "change": change,
168        "pct_change": pct_change,
169        "expected_reduction_factor": 2**order,
170    }

def chronological_split( table: pandas.DataFrame, *, time_col: str = 'collecteddate', train_frac: float = 0.8) -> tuple[pandas.DataFrame, pandas.DataFrame, pandas.Timestamp]: 
24def chronological_split(
25    table: pd.DataFrame,
26    *,
27    time_col: str = "collecteddate",
28    train_frac: float = 0.8,
29) -> tuple[pd.DataFrame, pd.DataFrame, pd.Timestamp]:
30    """Split a table into train/test sets by time, not at random.
31
32    All rows up to a single global cutoff timestamp (computed from
33    ``train_frac`` of the pooled, sorted timestamps across every room) go
34    to training; everything after goes to test. Using one global cutoff
35    (rather than splitting each room independently) keeps the train/test
36    boundary at the same wall-clock moment for every room, which is what
37    "can this model generalise to the future" should mean here.
38
39    Args:
40        table: Supervised table (e.g. from
41            :func:`occupancy.features.build_supervised_dataset`).
42        time_col: Column holding each row's timestamp.
43        train_frac: Fraction of rows (by time, not room-balanced count)
44            assigned to training.
45
46    Returns:
47        ``(train, test, cutoff_time)``.
48
49    Raises:
50        ValueError: If ``train_frac`` is not strictly between 0 and 1, or
51            the split would leave either set empty.
52    """
53    if not 0.0 < train_frac < 1.0:
54        raise ValueError(f"train_frac must be in (0, 1), got {train_frac}")
55
56    sorted_table = table.sort_values(time_col).reset_index(drop=True)
57    cutoff_idx = int(len(sorted_table) * train_frac)
58    if cutoff_idx == 0 or cutoff_idx >= len(sorted_table):
59        raise ValueError(
60            f"train_frac={train_frac} produced an empty train or test set "
61            f"for {len(sorted_table)} rows."
62        )
63
64    cutoff_time = sorted_table.loc[cutoff_idx, time_col]
65    train = sorted_table.iloc[:cutoff_idx].reset_index(drop=True)
66    test = sorted_table.iloc[cutoff_idx:].reset_index(drop=True)
67    return train, test, cutoff_time

def find_unbroken_intervals(series: pandas.Series) -> pandas.DataFrame: 
25def find_unbroken_intervals(series: pd.Series) -> pd.DataFrame:
26    """Find runs of consecutive non-missing values on a regular time grid.
27
28    Args:
29        series: A regularly binned series (e.g. the output of
30            :func:`occupancy.co2_regression.co2_series`), where missing
31            bins are ``NaN``.
32
33    Returns:
34        A DataFrame with one row per unbroken interval and columns
35        ``start``, ``end`` (index labels of the first and last bin) and
36        ``n_bins``, in time order. Empty if ``series`` has no values.
37    """
38    present = series.notna()
39    run_id = (present != present.shift()).cumsum()
40    runs = (
41        series.index.to_series()[present]
42        .groupby(run_id[present])
43        .agg(start="min", end="max", n_bins="size")
44    )
45    return runs.reset_index(drop=True)

def lagged_correlation( occupied: pandas.Series, variable: pandas.Series, max_lag_bins: int = 4, use_diff: bool = True, min_overlap_bins: int = 20) -> dict: 
 76def lagged_correlation(
 77    occupied: pd.Series,
 78    variable: pd.Series,
 79    max_lag_bins: int = DEFAULT_MAX_LAG_BINS,
 80    use_diff: bool = True,
 81    min_overlap_bins: int = MIN_OVERLAP_BINS,
 82) -> dict:
 83    """Find the strongest correlation between occupancy and a variable across a small lag search.
 84
 85    Args:
 86        occupied: Binary occupancy series (0.0/1.0), time-indexed, as
 87            produced by :func:`occupancy.preprocessing.resample_occupancy`
 88            for one room.
 89        variable: An environmental variable series, time-indexed on the
 90            same grid, as produced by :func:`resample_env_variable`.
 91        max_lag_bins: Search lags from ``-max_lag_bins`` to
 92            ``+max_lag_bins`` (in grid bins). A positive lag means the
 93            variable is shifted to *follow* occupancy (sensor reacts after
 94            the fact); negative means it *leads*.
 95        use_diff: If True (default), correlate against the variable's
 96            first difference (rate of change) rather than its raw level --
 97            see the module docstring for why this matters.
 98        min_overlap_bins: Minimum overlapping non-missing bins required to
 99            trust a candidate lag (default :data:`MIN_OVERLAP_BINS`,
100            appropriate for the real dataset's scale -- tests on small
101            hand-built series should pass a smaller value).
102
103    Returns:
104        A dict with ``best_lag`` (int or None if no lag had enough
105        overlapping data), ``correlation`` (float, Pearson r, NaN if
106        undetermined), and ``n_bins`` (overlapping non-missing bins used
107        for the best lag).
108    """
109    series = variable.diff() if use_diff else variable
110
111    candidates = []
112    for lag in range(-max_lag_bins, max_lag_bins + 1):
113        # shift(-lag) aligns occupied[t] with variable[t + lag]: a positive
114        # lag compares occupancy *now* to the variable *lag bins later*,
115        # i.e. "the variable follows occupancy" -- matching the docstring.
116        shifted = series.shift(-lag)
117        aligned = pd.concat(
118            [occupied.rename("occupied"), shifted.rename("variable")], axis=1
119        ).dropna()
120        if len(aligned) < min_overlap_bins:
121            continue
122        if aligned["occupied"].nunique() < 2 or aligned["variable"].nunique() < 2:
123            continue  # a constant series has an undefined correlation
124        corr = aligned["occupied"].corr(aligned["variable"])
125        candidates.append((lag, corr, len(aligned)))
126
127    if not candidates:
128        return {"best_lag": None, "correlation": float("nan"), "n_bins": 0}
129
130    best_lag, best_corr, n = max(candidates, key=lambda c: abs(c[1]))
131    return {"best_lag": best_lag, "correlation": best_corr, "n_bins": n}
